4. Breaking API 变更 — 6项

背景: OceanSim 原为 Isaac Sim 4.5 编写。Isaac Sim 6.0 重命名或移除了多个 API。不修复这些变更，扩展会因 ImportError 或 AttributeError 无法加载。
#APIIsaac 4.5Isaac 6.0影响文件
1模块注册 隐式自动发现 需在 extension.toml 中显式声明 [[python.module]] extension.toml
2依赖名称 omni.isaac.ui, omni.isaac.core isaacsim.gui.components, isaacsim.core.api, isaacsim.examples.extension extension.toml
3语义标注 add_update_semantics(prim, semantic_label, type_label) add_labels(prim, labels, instance_name) ui_builder.py
4PhysX 接口 acquire_physx_interface() get_physx_interface() extension.py (×2)
5UI 导入路径 omni.isaac.ui.element_wrappers, omni.isaac.ui.menu isaacsim.gui.components, isaacsim.gui.menu extension.py (×2)
6菜单项工厂 make_menu_item_description(ext_id, name, fn) MenuItemDescription(name=name, onclick_fn=fn) extension.py (×2)

verify_all_demos.py — 主验证脚本 (324行)
#!/usr/bin/env python3
"""OceanSim 在 Isaac Sim 6 上的全面验证."""
import sys, os, json
sys.stderr = open('/dev/null', 'w')
os.chdir(os.path.dirname(os.path.abspath(__file__)))

RESULTS = {}
def log(name, status, detail=""):
    RESULTS[name] = {"status": status, "detail": detail}
    tag = "PASS" if status else "FAIL"
    print(f"[{tag}] {name}: {detail}")

# 阶段1: Demo 数据文件 (无需 Isaac Sim)
import numpy as np
from PIL import Image

with open("demo/demo_waypoints.txt", "r") as f: lines = f.readlines()
waypoints = [[float(v) for v in l.strip().split()] for l in lines if len(l.strip().split()) == 7]
log("demo_waypoints.txt", len(waypoints) > 0, f"{len(waypoints)} waypoints loaded")

img = Image.open("demo/demo_rgb.png").convert("RGBA")
arr = np.array(img)
log("demo_rgb.png", arr.ndim == 3 and arr.shape[2] == 4, f"shape={arr.shape}")

depth = np.load("demo/demo_depth.npy")
log("demo_depth.npy", depth.ndim == 2, f"shape={depth.shape}, range=[{depth.min():.2f}, {depth.max():.2f}]")

# 阶段2: UW_render 核心
import warp as wp; wp.init()
from isaacsim.oceansim.utils.UWrenderer_utils import UW_render
demo_rgba = wp.array(data=arr, dtype=wp.uint8, ndim=3)
demo_depth = wp.array(data=depth, dtype=wp.float32, ndim=2)
uw_image = wp.zeros_like(demo_rgba)
param = [0.0, 0.31, 0.24, 0.05, 0.05, 0.2, 0.05, 0.05, 0.05]
wp.launch(dim=np.flip([arr.shape[1], arr.shape[0]]), kernel=UW_render,
    inputs=[demo_rgba, demo_depth, wp.vec3f(*param[0:3]), wp.vec3f(*param[6:9]), wp.vec3f(*param[3:6])],
    outputs=[uw_image])
result = uw_image.numpy()
log("UW_render kernel", result.ndim == 3 and result.shape[2] == 4,
    f"output shape={result.shape}, mean_rgb={result[:,:,:3].mean():.1f}")

# 阶段3-6: Isaac Sim 6
from isaacsim import SimulationApp
app = SimulationApp({"headless": True})
import omni.kit.app
mgr = omni.kit.app.get_app().get_extension_manager()
mgr.set_extension_enabled_immediate("OceanSim", True)

from isaacsim.oceansim.utils.assets_utils import get_oceansim_assets_path
from isaacsim.core.utils.stage import create_new_stage, add_reference_to_stage
from isaacsim.core.utils.prims import get_prim_at_path
from isaacsim.core.prims import SingleRigidPrim, SingleGeometryPrim
from isaacsim.core.utils.rotations import euler_angles_to_quat
from pxr import PhysxSchema

assets_path = get_oceansim_assets_path()
log("OceanSim assets", os.path.isdir(assets_path), assets_path)

create_new_stage()
rob = add_reference_to_stage(usd_path=assets_path+"/Bluerov/BROV_low.usd", prim_path="/World/rob")
rob_api = PhysxSchema.PhysxRigidBodyAPI.Apply(get_prim_at_path("/World/rob"))
rob_api.CreateDisableGravityAttr(True)
rob_api.GetLinearDampingAttr().Set(10.0)
rob_api.GetAngularDampingAttr().Set(10.0)
SingleGeometryPrim(prim_path="/World/rob", collision=True)
SingleRigidPrim(prim_path="/World/rob", mass=5.0, translation=np.array([-2.0, 0.0, -0.8]))
log("BlueROV robot loaded", True, assets_path+"/Bluerov/BROV_low.usd")

# 传感器测试
from isaacsim.oceansim.sensors.BarometerSensor import BarometerSensor
baro = BarometerSensor(prim_path="/World/rob/Baro", water_surface_z=1.43389)
pressure = baro.get_pressure()
log("BarometerSensor", isinstance(pressure, float) and pressure > 0, f"pressure={pressure:.1f} Pa")

from isaacsim.oceansim.sensors.DVLsensor import DVLsensor
dvl = DVLsensor(max_range=10)
dvl.attachDVL(rigid_body_path="/World/rob", translation=np.array([0, 0, -0.1]))
log("DVLsensor", len(dvl.get_beam_paths()) == 4, f"beams={len(dvl.get_beam_paths())}")

from isaacsim.oceansim.sensors.UW_Camera import UW_Camera
cam = UW_Camera(prim_path="/World/rob/UW_camera", resolution=[640, 480], translation=np.array([0.3, 0.0, 0.1]))
cam.set_focal_length(2.1); cam.set_clipping_range(0.1, 100)
log("UW_Camera", cam.get_resolution() == [640, 480], f"resolution={cam.get_resolution()}")

from isaacsim.oceansim.sensors.ImagingSonarSensor import ImagingSonarSensor
sonar = ImagingSonarSensor(prim_path="/World/rob/sonar", translation=np.array([0.3, 0.0, 0.3]),
    orientation=euler_angles_to_quat(np.array([0.0, 45, 0.0]), degrees=True),
    range_res=0.005, angular_res=0.25, hori_res=4000)
log("ImagingSonarSensor", sonar.get_range() is not None and sonar.get_fov() is not None,
    f"range={sonar.get_range()}, fov={sonar.get_fov()}")

# 场景循环
from isaacsim.oceansim.modules.SensorExample_python.scenario import MHL_Sensor_Example_Scenario
scenario = MHL_Sensor_Example_Scenario()
scenario.setup_scenario(rob=rob, sonar=None, cam=None, DVL=None, baro=baro, ctrl_mode="No control")
for i in range(5): scenario.update_scenario(1.0 / 60.0)
log("Sensor Example scenario", isinstance(scenario._baro_reading, float),
    f"baro_reading={scenario._baro_reading:.1f} Pa after 5 steps")
scenario.teardown_scenario()

# 航点跟踪
scenario2 = MHL_Sensor_Example_Scenario()
scenario2.setup_scenario(rob=rob, sonar=None, cam=None, DVL=None, baro=None, ctrl_mode="Waypoints")
scenario2.setup_waypoints(waypoint_path="demo/demo_waypoints.txt", default_waypoint_path="demo/demo_waypoints.txt")
initial_pos = rob.GetAttribute('xformOp:translate').Get()
for i in range(10): scenario2.update_scenario(1.0 / 60.0)
final_pos = rob.GetAttribute('xformOp:translate').Get()
scenario2.teardown_scenario()
moved = (abs(final_pos[0]-initial_pos[0])>0.01 or abs(final_pos[1]-initial_pos[1])>0.01 or abs(final_pos[2]-initial_pos[2])>0.01)
log("Waypoint following", moved, f"initial={initial_pos}, final={final_pos}")

# 颜色选择器 (3种水型)
all_ok = True
for name, p in [("clear",[0,.1,.05,.02,.02,.1,.02,.02,.02]), ("turbid",[0,.31,.24,.05,.05,.2,.05,.05,.05]), ("deep",[0,.05,.1,.01,.01,.05,.08,.04,.02])]:
    uw = wp.zeros_like(demo_rgba)
    wp.launch(dim=np.flip([arr.shape[1], arr.shape[0]]), kernel=UW_render,
        inputs=[demo_rgba, demo_depth, wp.vec3f(*p[0:3]), wp.vec3f(*p[6:9]), wp.vec3f(*p[3:6])], outputs=[uw])
    if uw.numpy().ndim != 3 or uw.numpy().shape[2] != 4: all_ok = False
log("Colorpicker UW_render", all_ok, f"tested 3 water types")

# 汇总
total = len(RESULTS); passed = sum(1 for v in RESULTS.values() if v["status"]); failed = total - passed
with open("verification_results.json", "w") as f:
    json.dump({"total":total, "passed":passed, "failed":failed, "results":RESULTS}, f, indent=2)
app.close()
sys.exit(0 if failed == 0 else 1)
